cmake_minimum_required(VERSION 3.20)
project(CheeseBot LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(policy-test tests/policy.cpp)
add_test(NAME decision-policy COMMAND policy-test)

option(BUILD_BWAPI_CLIENT "Build the Windows BWAPI 4.4.0 client executable" OFF)
if(BUILD_BWAPI_CLIENT)
  if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "The native client requires Windows, MSVC, and a matching Win32 BWAPI 4.4.0 build.")
  endif()
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Configure the client with -A Win32.")
  endif()
  set(BWAPI_DIR "$ENV{BWAPI_DIR}" CACHE PATH "BWAPI 4.4.0 root")
  find_path(BWAPI_INCLUDE_DIR BWAPI.h HINTS "${BWAPI_DIR}/include" REQUIRED)
  set(BWAPI_LIBRARY "" CACHE FILEPATH "BWAPILIB.lib built with the same compiler and configuration")
  set(BWAPI_CLIENT_LIBRARY "" CACHE FILEPATH "BWAPIClient.lib built with the same compiler and configuration")
  if(NOT EXISTS "${BWAPI_LIBRARY}" OR NOT EXISTS "${BWAPI_CLIENT_LIBRARY}")
    message(FATAL_ERROR "Set BWAPI_LIBRARY and BWAPI_CLIENT_LIBRARY to matching locally built .lib files. See README.md.")
  endif()
  add_executable(CheeseBot StarCraftCannonRushBot/main.cpp StarCraftCannonRushBot/StarterBot.cpp StarCraftCannonRushBot/Tools.cpp StarCraftCannonRushBot/MapTools.cpp)
  target_include_directories(CheeseBot PRIVATE "${BWAPI_INCLUDE_DIR}")
  target_link_libraries(CheeseBot PRIVATE "${BWAPI_LIBRARY}" "${BWAPI_CLIENT_LIBRARY}")
endif()
